    async function visitExample(data) {

      status("正在前往：" + data.name);


      const location = Cesium.Cartographic.fromDegrees(

        data.lon, data.lat

      );


      let height = 0;


      try {

        const result = await Cesium.sampleTerrainMostDetailed(

          viewer.terrainProvider,

          [location]

        );


        height = Number.isFinite(result[0].height)

          ? result[0].height

          : 0;

      } catch (error) {

        console.warn("未取得詳細地形高度。", error);

        height = viewer.scene.globe.getHeight(location) || 0;

      }


      const target = Cesium.Cartesian3.fromDegrees(

        data.lon, data.lat, Math.max(0, height)

      );


      const range =

        data.word === "山" ? 5000 :

        data.word === "湖" ? 1000 : 3500;


      viewer.camera.lookAt(

        target,

        new Cesium.HeadingPitchRange(

          Cesium.Math.toRadians(20),

          Cesium.Math.toRadians(-40),

          range

        )

      );


      // 解除環繞鎖定，恢復一般地圖操作。

      viewer.camera.lookAtTransform(

        Cesium.Matrix4.IDENTITY

      );


      status("請點擊彩色圓點：" + data.name);

    }

    // ===== 4. 匯入 GeoJSON =====

    // 每個 Feature 的 properties.word 必須是：

    // 山、江、河、湖、海之一。

    async function importGeoJSON(file) {

      let source;


      try {

        status("正在匯入資料……");


        const json = JSON.parse(await file.text());


        if (

          json.type !== "FeatureCollection" ||

          !Array.isArray(json.features) ||

          json.features.length === 0

        ) {

          throw new Error(

            "請提供非空的 GeoJSON FeatureCollection。"

          );

        }


        for (const feature of json.features) {

          const word = feature.properties?.word;


          if (

            !Object.prototype.hasOwnProperty.call(

              vocabulary, word

            )

          ) {

            throw new Error(

              "每個地物的 properties.word " +

              "必須是山、江、河、湖或海。"

            );

          }


          if (!feature.geometry) {

            throw new Error("地物不能缺少 geometry。");

          }

        }


        source = await Cesium.GeoJsonDataSource.load(json, {

          clampToGround: true,

          stroke: Cesium.Color.DEEPSKYBLUE,

          strokeWidth: 10,

          fill: Cesium.Color.DEEPSKYBLUE.withAlpha(0.35)

        });


        await viewer.dataSources.add(source);


        for (const entity of source.entities.values) {

          const properties = entity.properties?.getValue(

            viewer.clock.currentTime

          );


          if (!properties) continue;


          const word = properties.word;

          const base = vocabulary[word];


          const data = {

            word,

            name: properties.name || "匯入的學習地物",

            meaning: properties.meaning || base.meaning,

            sentence: properties.sentence || base.sentence

          };


          lessons.set(entity, data);


          const color = Cesium.Color.fromCssColorString(

            base.color

          );


          // 河流線：加寬以方便點擊。

          if (entity.polyline) {

            entity.polyline.width = 10;

            entity.polyline.material = color;

          }


          // 湖泊或海域面：點擊面內即可觸發。

          if (entity.polygon) {

            entity.polygon.material = color.withAlpha(0.4);

          }


          if (entity.billboard) {

            entity.billboard.color = color;

          }

        }


        await viewer.flyTo(source);


        status(

          "已匯入 " + json.features.length +

          " 個地物。可點擊其標記、線或面。"

        );

      } catch (error) {

        console.error(error);


        if (source && viewer.dataSources.contains(source)) {

          for (const entity of source.entities.values) {

            lessons.delete(entity);

          }


          viewer.dataSources.remove(source, true);

        }


        status("匯入失敗：" + error.message);

      } finally {

        $("file").value = "";

      }

    }


    // ===== 5. 朗讀 =====

    function speakLesson() {

      if (!currentLesson) return;


      if (!("speechSynthesis" in window)) {

        status("這個瀏覽器不支援語音朗讀。");

        return;

      }


      window.speechSynthesis.cancel();


      const utterance = new SpeechSynthesisUtterance(

        currentLesson.word + "。" + currentLesson.sentence

      );


      utterance.lang = "zh-CN";

      utterance.rate = 0.8;


      const voice = window.speechSynthesis.getVoices().find(

        item => /^zh[-_]CN$/i.test(item.lang)

      );


      if (voice) {

        utterance.voice = voice;

      }


      utterance.onerror = () => {

        status(

          "朗讀失敗：請檢查裝置是否安裝中文語音，" +

          "或改用其他瀏覽器。"

        );

      };


      window.speechSynthesis.speak(utterance);

    }


    // ===== 6. 啟動地圖 =====

    async function init() {

      try {

        if (typeof Cesium === "undefined") {

          throw new Error(

            "Cesium 程式庫未載入，請檢查網絡。"

          );

        }


        if (

          !CESIUM_TOKEN ||

          CESIUM_TOKEN.includes("在這裏")

        ) {

          throw new Error(

            "請先在程式內填入你的 Cesium Token。"

          );

        }


        Cesium.Ion.defaultAccessToken = CESIUM_TOKEN;


        status("正在載入三維地形及影像……");


        const [terrainProvider, imageryProvider] =

          await Promise.all([

            Cesium.createWorldTerrainAsync(),

            Cesium.createWorldImageryAsync()

          ]);


        viewer = new Cesium.Viewer("map", {

          terrainProvider,

          baseLayer: new Cesium.ImageryLayer(imageryProvider),

          animation: false,

          timeline: false,

          baseLayerPicker: false,

          geocoder: false,

          homeButton: false,

          sceneModePicker: false,

          navigationHelpButton: false,

          infoBox: false,

          selectionIndicator: false

        });


        // 使用實際地形比例，不額外誇大山高。

        viewer.scene.verticalExaggeration = 1;


        viewer.scene.globe.depthTestAgainstTerrain = true;


        // 不隱藏 Cesium 或影像供應商的署名。

        examples.forEach(addExample);

        setupClick();


        $("home").disabled = false;

        $("places").disabled = false;

        $("file").disabled = false;


        showHongKong();


        status(

          "地圖已啟動，共 " + examples.length +

          " 個學習示例。請選擇地點或點擊彩色圓點。"

        );

      } catch (error) {

        console.error(error);


        status(

          "載入失敗：" + error.message +

          " 請檢查 Token、資產權限及網絡。"

        );

      }

    }
